# Version ranges, not bare names. An unpinned deploy resolves to whatever was
# published the day it built, so a redeploy that changes nothing in this repo can
# still come up broken — and the release that breaks it is usually a major one
# that landed between two deploys. Lower bounds are what the code is written
# against; upper bounds stop the next major version from arriving unannounced.
#
# Verified together against Python 3.11.

fastapi>=0.115,<0.123
uvicorn>=0.30,<0.39
python-multipart>=0.0.9,<0.1

# Pinned explicitly even though FastAPI pulls it in: FastAPI accepts anything
# from 0.40 up, and GZipMiddleware only learned to leave text/event-stream alone
# in later releases. On an older one it compresses the chat stream into a single
# buffered blob, so answers appear all at once at the end instead of token by
# token — a regression that would show up only in a fresh deploy's dependency
# resolution, never in an existing environment.
starlette>=0.47,<0.50

# pandas 3 renamed the default string dtype from "object" to "str"; the loaders
# and trainers use the pandas type API rather than dtype-name matching so both
# 2.x and 3.x work. numpy stays below 2.0 for xgboost/scipy wheel compatibility.
pandas>=2.2,<3.1
numpy>=1.26,<2.0
scipy>=1.11,<2.0
scikit-learn>=1.4,<1.9

openpyxl>=3.1,<4.0

# xgboost-cpu on Linux, plain xgboost elsewhere. Both import as `xgboost`, so
# core/trainer.py needs no knowledge of which one is installed.
#
# The default Linux wheel depends on nvidia-nccl-cu12 and carries a GPU-enabled
# libxgboost.so — together 628 MB of the image, on an instance with no GPU and a
# 2 GB disk. The CPU build is the same library without the CUDA half. The marker
# is needed because xgboost-cpu publishes Linux wheels only, so a Windows or
# macOS dev machine installing from this file must still get the normal package.
xgboost-cpu>=2.1,<4.0; sys_platform == "linux"
xgboost>=2.0,<4.0; sys_platform != "linux"

reportlab>=4.0,<5.0
python-dotenv>=1.0,<2.0
anthropic>=0.40,<1.0
